Replit Agent Prompt: Build n8n-Powered Integration Marketplace for Loop9
Overview
Build a white-labeled Integration Marketplace in Loop9 powered by n8n (self-hosted workflow automation). Users can browse integrations, connect CRMs/ERPs with one click, and set up automated workflows—all within the Loop9 panel. n8n handles OAuth, field mapping, and data sync, while appearing as "Loop9 Integrations" to end users.

Architecture: Loop9 + n8n Integration
text
Loop9 (Frontend/UI)
    ↓
n8n (Self-hosted workflow engine)
    ↓
CRM/ERP APIs (Salesforce, HubSpot, Zoho, etc.)

✅ 400+ pre-built integrations via n8n

✅ 100% white-labeled (zero n8n branding visible)

✅ Self-hosted (full data control)

✅ Visual workflow builder

✅ Unlimited executions

✅ Cost: ~$30-50/month (vs $1,500+ for alternatives)

2.2 Database Schema
Create these tables in Supabase:

sql
-- Integration apps catalog
CREATE TABLE integration_apps (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL,
  description TEXT,
  logo_url TEXT,
  category VARCHAR(50), -- 'crm', 'erp', 'marketing', 'sales'
  n8n_node_name VARCHAR(100), -- e.g., 'Salesforce', 'HubSpot'
  is_active BOOLEAN DEFAULT true,
  is_popular BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT NOW()
);

-- User's connected integrations (n8n workflows)
CREATE TABLE user_integrations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id),
  app_id UUID REFERENCES integration_apps(id),
  n8n_workflow_id VARCHAR(100), -- n8n workflow ID
  n8n_webhook_url TEXT, -- Unique webhook URL
  status VARCHAR(20) DEFAULT 'active', -- 'active', 'inactive', 'error'
  workflow_config JSONB, -- Field mappings and settings
  last_sync_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(user_id, app_id)
);

-- Integration sync logs
CREATE TABLE integration_sync_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  integration_id UUID REFERENCES user_integrations(id),
  n8n_execution_id VARCHAR(100),
  sync_type VARCHAR(50), -- 'call_completed', 'lead_created'
  status VARCHAR(20), -- 'success', 'failed', 'pending'
  records_synced INTEGER DEFAULT 0,
  error_message TEXT,
  payload JSONB,
  synced_at TIMESTAMP DEFAULT NOW()
);

3.2 Create n8n Workflow Programmatically
When user clicks "Connect Salesforce":

javascript
// Backend: /api/integrations/salesforce/connect
async function connectSalesforce(userId, appSlug) {
  // 1. Create workflow in n8n via API
  const workflow = await createN8nWorkflow({
    name: `Loop9 → Salesforce (User ${userId})`,
    nodes: [
      {
        // Webhook trigger node
        type: 'n8n-nodes-base.webhook',
        parameters: {
          path: `loop9-${userId}-${appSlug}`,
          method: 'POST'
        },
        name: 'Loop9 Webhook'
      },
      {
        // Salesforce node
        type: 'n8n-nodes-base.salesforce',
        parameters: {
          resource: 'lead',
          operation: 'create',
          // Map Loop9 fields to Salesforce
          additionalFields: {
            firstName: '={{$json["contact_name"].split(" ")[0]}}',
            lastName: '={{$json["contact_name"].split(" ")[1]}}',
            company: '={{$json["customer_name"]}}',
            phone: '={{$json["phone_number"]}}',
            description: '={{$json["call_summary"]}}'
          }
        },
        name: 'Create Salesforce Lead',
        credentials: {
          // User will authenticate via n8n OAuth
          salesforceOAuth2Api: 'salesforce_credentials'
        }
      }
    ]
  });

  // 2. Get webhook URL
  const webhookUrl = workflow.webhookUrl;

  // 3. Save to database
  await saveIntegration({
    user_id: userId,
    app_slug: appSlug,
    n8n_workflow_id: workflow.id,
    n8n_webhook_url: webhookUrl,
    status: 'active'
  });

  // 4. Return OAuth URL for user to authenticate
  return {
    workflow_id: workflow.id,
    webhook_url: webhookUrl,
    auth_url: `https://integrations.loop9.com/oauth/salesforce?workflow_id=${workflow.id}`
  };
}
3.3 Trigger Workflows from Loop9
When a call completes in Loop9:

javascript
// Loop9: onCallCompleted event
async function onCallCompleted(callData, userId) {
  // Get user's active integrations
  const integrations = await getUserIntegrations(userId);

  for (const integration of integrations) {
    // Send data to n8n webhook
    try {
      const response = await fetch(integration.n8n_webhook_url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          event: 'call_completed',
          timestamp: new Date().toISOString(),
          data: {
            customer_name: callData.customerName,
            contact_name: callData.contactName,
            phone_number: callData.phoneNumber,
            contact_phone: callData.contactPhone,
            call_summary: callData.summary,
            case_notes: callData.notes,
            product_code: callData.productCode,
            erp_product_id: callData.erpProductId,
            call_duration: callData.duration,
            call_recording_url: callData.recordingUrl,
            lead_score: callData.leadScore
          }
        })
      });

      // Log success
      await logSync({
        integration_id: integration.id,
        n8n_execution_id: response.headers.get('x-n8n-execution-id'),
        status: 'success',
        records_synced: 1
      });
    } catch (error) {
      // Log failure
      await logSync({
        integration_id: integration.id,
        status: 'failed',
        error_message: error.message
      });
    }
  }
}
