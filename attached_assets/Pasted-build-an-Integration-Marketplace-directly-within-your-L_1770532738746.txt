2. OAuth 2.0 Authentication Flow
When user clicks "Connect" on any CRM/ERP app:

Step 1: User clicks "Connect Salesforce" button
Step 2: Modal opens with OAuth consent screen
Step 3: User is redirected to Salesforce login page
Step 4: User authorizes Loop9 to access their Salesforce data
Step 5: Salesforce redirects back to Loop9 with authorization code
Step 6: Loop9 exchanges code for access token + refresh token
Step 7: Tokens are securely stored in Loop9 database (encrypted)
Step 8: Success screen: "✓ Salesforce Connected Successfully"

CREATE TABLE integration_apps (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL,
  description TEXT,
  logo_url TEXT,
  category VARCHAR(50), -- 'crm', 'erp', 'marketing', 'sales', 'accounting'
  oauth_config JSONB, -- {client_id, auth_url, token_url, scopes}
  available_fields JSONB, -- List of fields this app supports
  is_active BOOLEAN DEFAULT true,
  is_popular BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT NOW()
);

-- User's connected integrations
CREATE TABLE user_integrations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id),
  app_id UUID REFERENCES integration_apps(id),
  access_token TEXT, -- Encrypted
  refresh_token TEXT, -- Encrypted
  token_expires_at TIMESTAMP,
  status VARCHAR(20) DEFAULT 'active', -- 'active', 'inactive', 'error'
  field_mapping JSONB, -- Maps Loop9 fields to CRM/ERP fields
  workflow_config JSONB, -- Enabled workflows and automation rules
  last_sync_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(user_id, app_id)
);

-- Integration sync logs
CREATE TABLE integration_sync_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  integration_id UUID REFERENCES user_integrations(id),
  sync_type VARCHAR(50), -- 'call_completed', 'lead_created', 'manual'
  status VARCHAR(20), -- 'success', 'failed', 'pending'
  records_synced INTEGER DEFAULT 0,
  error_message TEXT,
  payload JSONB,
  synced_at TIMESTAMP DEFAULT NOW()
);

6. OAuth 2.0 Implementation
For the OAuth flow, implement:

javascript
// 1. Generate OAuth URL
function generateOAuthUrl(app, userId) {
  const state = generateSecureState(userId, app.slug);
  const params = new URLSearchParams({
    client_id: app.oauth_config.client_id,
    redirect_uri: `${process.env.BASE_URL}/api/integrations/callback`,
    response_type: 'code',
    scope: app.oauth_config.scopes.join(' '),
    state: state
  });
  return `${app.oauth_config.auth_url}?${params}`;
}

// 2. Handle OAuth callback
async function handleOAuthCallback(code, state, appSlug) {
  // Verify state token
  const { userId } = verifyState(state);
  
  // Exchange code for tokens
  const tokens = await exchangeCodeForTokens(appSlug, code);
  
  // Store encrypted tokens in database
  await storeIntegration(userId, appSlug, tokens);
  
  // Redirect to success page
  return `/app/integrations/${appSlug}?status=connected`;
}

async function onCallCompleted(callData, userId) {
  // Get user's active integrations
  const integrations = await getUserIntegrations(userId);
  
  for (const integration of integrations) {
    // Map Loop9 fields to CRM fields
    const mappedData = mapFields(callData, integration.field_mapping);
    
    // Send to CRM via API
    try {
      await syncToCRM(integration, mappedData);
      await logSync(integration.id, 'call_completed', 'success');
    } catch (error) {
      await logSync(integration.id, 'call_completed', 'failed', error.message);
    }
  }
}
